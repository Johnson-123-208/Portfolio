            const container = document.getElementById('particles');
            const particleCount = window.innerWidth < 768 ? 30 : 50; // Fewer particles on mobile

        // Neural Network Background
        function createNeuralNetwork() {
            if (window.innerWidth < 768) return; // Skip on mobile for performance
            
            const container = document.getElementById('neuralBg');
            const nodeCount = 20;
            const nodes = [];
            
            // Create nodes
            for (let i = 0; i < nodeCount; i++) {
                const node = document.createElement('div');
                node.className = 'node';
                node.style.left = Math.random() * 100 + '%';
                node.style.top = Math.random() * 100 + '%';
                container.appendChild(node);
                nodes.push(node);
            }
            
            // Create connections
            for (let i = 0; i < nodes.length; i++) {
                for (let j = i + 1; j < nodes.length; j++) {
                    if (Math.random() < 0.1) { // 10% chance of connection
                        const connection = document.createElement('div');
                        connection.className = 'connection';
                        
                        const node1Rect = nodes[i].getBoundingClientRect();
                        const node2Rect = nodes[j].getBoundingClientRect();
                        
                        const x1 = node1Rect.left + node1Rect.width / 2;
                        const y1 = node1Rect.top + node1Rect.height / 2;
                        const x2 = node2Rect.left + node2Rect.width / 2;
                        const y2 = node2Rect.top + node2Rect.height / 2;
                        
                        const length = Math.sqrt((x2 - x1) ** 2 + (y2 - y1) ** 2);
                        const angle = Math.atan2(y2 - y1, x2 - x1) * 180 / Math.PI;
                        
                        connection.style.width = length + 'px';
                        connection.style.left = x1 + 'px';
                        connection.style.top = y1 + 'px';
                        connection.style.transform = `rotate(${angle}deg)`;
                        connection.style.animationDelay = Math.random() * 3 + 's';
                        
                        container.appendChild(connection);
                    }
                }
            }
        }
